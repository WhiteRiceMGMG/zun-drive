	○○要求判定関数
	{
	    if ( ○○最新値 == ○○ )  /* 優先度考慮 */
	    {
	        ○○要求 ← ○○要求A
	    }

	    {
	        ○○要求 ← ○○要求B
	    }

	    {
	        ○○要求 ← ○○要求C

	            ○○許可_ワーク ← ON
	            break;
	        case ○○要求B:
	            if ( ( ○○ < ○○ )
	              && ( ○○ > ○○ )
	              && ( ○○ < ○○ ) )
	            {
	                ○○許可_ワーク ← ON
	            }
	            break;
	        case ○○要求C:
	            if ( ○○最新値 == ○○ ) /* ○○からの○○は常に許可 */
	            {
	                ○○許可_ワーク ← ON
	            }

	            {
	                if ( ( ○○ > ○○ )
	                  && ( ○○ > ○○ )

	                {
